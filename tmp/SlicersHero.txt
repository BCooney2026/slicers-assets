function useFonts() {
    React.useEffect(() => {
        if (document.getElementById("slicers-gf")) return
        const link = document.createElement("link")
        link.id = "slicers-gf"
        link.rel = "stylesheet"
        link.href =
            "https://fonts.googleapis.com/css2?family=Playfair+Display:ital,wght@0,700;0,900;1,700;1,900&family=Lora:wght@400;500;600&family=DM+Sans:wght@400;500;700&family=Dancing+Script:wght@700&display=swap"
        document.head.appendChild(link)
    }, [])
}

const FALLBACKS = [
    "linear-gradient(145deg,#c9b99a,#ddd0b8)",
    "linear-gradient(145deg,#bfad93,#d4c5a9)",
    "linear-gradient(145deg,#b5a188,#cfc0a4)",
]
function photoSrc(src?: string, i = 0) {
    return src && src.trim() ? src : FALLBACKS[i % FALLBACKS.length]
}
function isGrad(v: string) {
    return v.startsWith("linear-gradient") || v.startsWith("radial-gradient")
}

}) {
    return (
        <div
            style={{
                fontFamily: SANS,

}

function RotatingPhotoCard({
    sources,
    label,
    index,
    style = {},
}: {
    sources: (string | undefined)[]
    label: string
    index: number
    style?: React.CSSProperties
}) {
    const usableSources = React.useMemo(
        () =>
            sources.filter((src): src is string => Boolean(src && src.trim())),
        [sources]
    )

    const [activeIndex, setActiveIndex] = React.useState(0)

    React.useEffect(() => {
        if (usableSources.length <= 1) return
        // Respect "reduce motion": keep the first photo, no rotation.
        if (
            typeof window !== "undefined" &&
            window.matchMedia &&
            window.matchMedia("(prefers-reduced-motion: reduce)").matches
        )
            return

        let timeoutId: number | null = null

        function randomDelay() {
            return 3000 + Math.floor(Math.random() * 2000)
        }

        function getNextIndex(current: number) {
            if (usableSources.length <= 1) return 0

            let next = Math.floor(Math.random() * usableSources.length)

            if (next === current) {
                next = (next + 1) % usableSources.length
            }

            return next
        }

        function scheduleNextSwitch(delay?: number) {
            const nextDelay = delay ?? 3000 + Math.floor(Math.random() * 5000)

            timeoutId = window.setTimeout(() => {
                setActiveIndex((current) => getNextIndex(current))

                // Generate a completely new random delay every cycle
                scheduleNextSwitch(3000 + Math.floor(Math.random() * 5000))
            }, nextDelay)
        }

        scheduleNextSwitch(3000 + index * 2200)

        return () => {
            if (timeoutId !== null) {
                window.clearTimeout(timeoutId)
            }
        }
    }, [usableSources.length])

    const img =
        usableSources[activeIndex] || FALLBACKS[index % FALLBACKS.length]

    return (
        <div
            style={{
                position: "relative",
                borderRadius: 8,
                overflow: "hidden",
                background: isGrad(img) ? img : "#c9b99a",
                boxShadow: "0 6px 22px rgba(15,11,8,.20)",
                ...style,
            }}
        >
            {usableSources.length > 0 ? (
                <img
                    key={`${label}-${activeIndex}-${img}`}
                    src={img}
                    alt={label}
                    style={{
                        position: "absolute",
                        inset: 0,
                        width: "100%",
                        height: "100%",
                        objectFit: "cover",
                        transform: "scale(1.03)",
                    }}
                />
            ) : !isGrad(img) ? (
                <img
                    src={img}
                    alt={label}
                    style={{
                        position: "absolute",
                        inset: 0,
                        width: "100%",
                        height: "100%",
                        objectFit: "cover",
                    }}
                />
            ) : null}

            <div
                style={{
                    position: "absolute",
                    inset: 0,
                    background:
                        "linear-gradient(to bottom,rgba(15,11,8,0) 40%,rgba(15,11,8,.62))",
                }}
            />
            <div
                style={{
                    position: "absolute",
                    left: 13,
                    bottom: 13,
                    fontFamily: SANS,
                    fontWeight: 700,
                    fontSize: 10,
                    letterSpacing: "0.16em",
                    textTransform: "uppercase",
                    color: "rgba(255,255,255,.80)",
                }}
            >
                {label}
            </div>
        </div>
    )
}

// Fisher-Yates shuffle — runs once at module load, different order every page visit
function shuffleArray<T>(arr: T[]): T[] {
    const a = [...arr]
    for (let i = a.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1))
        ;[a[i], a[j]] = [a[j], a[i]]
    }
    return a
}

const REVIEWS = shuffleArray(ALL_REVIEWS)

function ReviewCarousel() {
    const VISIBLE = 5
    const GAP = 14
    const HOLD_MS = 5000
    const SLIDE_MS = 650
    const CARD_HEIGHT = 258

    const outerRef = React.useRef<HTMLDivElement>(null)
    const timeoutRef = React.useRef<number | null>(null)
    const [cardW, setCardW] = React.useState(0)
    const [startIndex, setStartIndex] = React.useState(0)
    const [offset, setOffset] = React.useState(0)
    const [animating, setAnimating] = React.useState(false)

    React.useEffect(() => {
        function measure() {
            if (!outerRef.current) return
            const available = outerRef.current.offsetWidth
            const visibleCount =
                available < 720
                    ? 1
                    : available < 980
                      ? 2
                      : available < 1180
                        ? 3
                        : VISIBLE
            const w = Math.floor(
                (available - GAP * (visibleCount - 1)) / visibleCount
            )
            if (w > 0) setCardW(w)
        }

        measure()
        window.addEventListener("resize", measure)
        return () => window.removeEventListener("resize", measure)
    }, [])

    React.useEffect(() => {
        if (cardW <= 0) return

        const step = cardW + GAP
        const intervalId = window.setInterval(() => {
            setAnimating(true)
            setOffset(step)

            timeoutRef.current = window.setTimeout(() => {
                setAnimating(false)
                setOffset(0)
                setStartIndex((current) => (current + 1) % REVIEWS.length)
            }, SLIDE_MS + 20)
        }, HOLD_MS)

        return () => {
            window.clearInterval(intervalId)
            if (timeoutRef.current !== null)
                window.clearTimeout(timeoutRef.current)
        }
    }, [cardW])

    const visibleCards = Array.from({ length: VISIBLE + 1 }, (_, i) => {
        const reviewIndex = (startIndex + i) % REVIEWS.length
        return {
            id: `${startIndex}-${i}-${REVIEWS[reviewIndex].name}`,
            review: REVIEWS[reviewIndex],
        }
    })

    return (
        <div
            ref={outerRef}
            style={{
                width: "100%",
                overflowX: "hidden",
                overflowY: "visible",
                minHeight: CARD_HEIGHT,
            }}

// MAIN
// V23 — MOBILE FIX: specialties now flow, nav height responsive, hero hours corrected
// ─────────────────────────────────────────────
// ─────────────────────────────────────────────
// STOCK FEED: one fetch of the Apps Script JSON, shared by the Weekly Hero
// and the Fresh Today band. Refreshes every 60s.
//
// Staleness rule: if the last GOOD response is more than 5 minutes old
// (the feed is down, the tablet's Sheet is unreachable, the visitor's
// connection dropped), "fresh" goes false and the page stops showing any
// availability or "On the counter now" claims until a good response comes
// back. An old promotion can never keep claiming it is available.
//
// preview: a Limited Time sandwich name to show in the hero even if it is
// off or scheduled (used only on /home-wip via the "Preview LTO" control).
// ─────────────────────────────────────────────
const STALE_MS = 5 * 60 * 1000
// Speed: the last good feed is remembered in the visitor's browser for up
// to 6 hours, so a returning visitor sees the LTO hero instantly while the
// live check runs (the live answer replaces it a second or two later).
// First-time visitors get a same-size placeholder instead of a jump.
const FEED_CACHE_KEY = "slicersFeedV1"
const FEED_CACHE_MAX_MS = 6 * 60 * 60 * 1000
const FIRST_LOAD_GIVE_UP_MS = 8000

function readFeedCache(): StockItem[] | null {
    try {
        const raw = window.localStorage.getItem(FEED_CACHE_KEY)
        if (!raw) return null
        const c = JSON.parse(raw)
        if (!c || !Array.isArray(c.items)) return null
        if (Date.now() - Number(c.at || 0) > FEED_CACHE_MAX_MS) return null
        return c.items
    } catch (e) {
        return null
    }
}

function writeFeedCache(items: StockItem[]) {
    try {
        window.localStorage.setItem(
            FEED_CACHE_KEY,
            JSON.stringify({ at: Date.now(), items })
        )
    } catch (e) {
        /* private mode or storage full: just skip it */
    }
}

// Warm up the connections to Google and the photo CDN as early as possible.
function preconnectOnce() {
    if (typeof document === "undefined") return
    if (document.getElementById("slicers-preconnect")) return
    ;[
        "https://script.google.com",
        "https://script.googleusercontent.com",
        "https://cdn.jsdelivr.net",
    ].forEach((href, i) => {
        const link = document.createElement("link")
        link.rel = "preconnect"
        link.href = href
        link.crossOrigin = "anonymous"
        if (i === 0) link.id = "slicers-preconnect"
        document.head.appendChild(link)
    })
}

function useStockFeed(enabled: boolean, preview?: string) {
    const [items, setItems] = React.useState<StockItem[]>([])
    const [cached, setCached] = React.useState<StockItem[] | null>(null)
    const [lastOk, setLastOk] = React.useState(0)
    const [loading, setLoading] = React.useState(true)
    const [now, setNow] = React.useState(() => Date.now())

    React.useEffect(() => {
        if (!enabled) return
        let cancelled = false
        const previewName = (preview || "").trim()
        preconnectOnce()
        if (!previewName) {
            const c = readFeedCache()
            if (c) setCached(c)
        }
        const giveUp = window.setTimeout(() => {
            if (!cancelled) setLoading(false)
        }, FIRST_LOAD_GIVE_UP_MS)

        function load() {
            const url =
                AVAIL_ENDPOINT +
                (AVAIL_ENDPOINT.indexOf("?") >= 0 ? "&" : "?") +
                "t=" +
                Date.now() +
                (previewName
                    ? "&preview=" + encodeURIComponent(previewName)
                    : "")
            fetch(url)
                .then((r) => r.json())
                .then((data) => {
                    if (cancelled) return
                    if (data && data.ok && Array.isArray(data.items)) {
                        setItems(data.items)
                        setLastOk(Date.now())
                        setLoading(false)
                        if (!previewName) writeFeedCache(data.items)
                    }
                })
                .catch(() => {
                    /* keep last good data; the staleness rule hides it */
                })
        }
        load()
        const fetchTimer = window.setInterval(load, 60000)
        const clockTimer = window.setInterval(() => setNow(Date.now()), 30000)
        return () => {
            cancelled = true
            window.clearTimeout(giveUp)
            window.clearInterval(fetchTimer)
            window.clearInterval(clockTimer)
        }
    }, [enabled, preview])

    // Live answer within the last 5 minutes: use it.
    if (lastOk > 0 && now - lastOk < STALE_MS) {
        return { items, ready: true, pending: false }
    }
    // Still waiting on the first live answer: use the browser's copy if any.
    if (enabled && loading && lastOk === 0 && cached) {
        return { items: cached, ready: true, pending: false }
    }
    // Waiting with nothing to show yet: the hero shows its placeholder.
    // Otherwise (feed down or stale): show no availability at all.
    return {
        items: [] as StockItem[],
        ready: false,
        pending: enabled && loading && lastOk === 0,
    }
}

// Visitor asked their device for less motion.
function usePrefersReducedMotion() {
    const [reduced, setReduced] = React.useState(false)
    React.useEffect(() => {
        if (typeof window === "undefined" || !window.matchMedia) return
        const mq = window.matchMedia("(prefers-reduced-motion: reduce)")
        const update = () => setReduced(mq.matches)
        update()
        if (mq.addEventListener) mq.addEventListener("change", update)
        return () => {
            if (mq.removeEventListener) mq.removeEventListener("change", update)
        }
    }, [])
    return reduced
}

// True on phone-width screens (matches the 760px phone breakpoint).
function useIsPhone() {
    const [phone, setPhone] = React.useState(false)
    React.useEffect(() => {
        if (typeof window === "undefined" || !window.matchMedia) return
        const mq = window.matchMedia("(max-width: 760px)")
        const update = () => setPhone(mq.matches)
        update()
        if (mq.addEventListener) mq.addEventListener("change", update)
        return () => {
            if (mq.removeEventListener) mq.removeEventListener("change", update)
        }
    }, [])
    return phone
}

// 04 DINNER IS HANDLED ($29 dinner specials)
// Day tabs (keyboard: arrow keys, Home, End) and one meal panel.
// Opens on today's dinner in Slicers' time zone. Monday shows an honest
// "closed today" note and opens on the next dinner, never calling it
// today's. A day with no special gets the same honest note.
// ─────────────────────────────────────────────
function dinnerLabel(day: string) {
    return day === "Sat / Sun" ? "Weekend" : day
}

function DinnerSpecials({
    dinners,
    todayKey,
    override,
    telHref,
    facebookUrl,
}: {
    dinners: DinnerItem[]
    todayKey: string // "Tuesday" ... "Sat / Sun", or "Closed"
    override?: string
    telHref: string
    facebookUrl: string
}) {
    const list = DINNER_DAYS.map((d) =>
        dinners.find((x) => x.day === d && x.show !== false)
    ).filter((x): x is DinnerItem => !!x)

    const hasToday = list.some((d) => d.day === todayKey)
    const initial =
        override && override !== "Auto" && list.some((d) => d.day === override)
            ? override
            : hasToday
              ? todayKey
              : list.length
                ? list[0].day || ""
                : ""

    const [sel, setSel] = React.useState(initial)
    React.useEffect(() => setSel(initial), [initial])
    const tabRefs = React.useRef<(HTMLButtonElement | null)[]>([])

    if (list.length === 0) return null
    const current = list.find((d) => d.day === sel) || list[0]
    const curDay = current.day || ""
    const isToday = curDay === todayKey
    const items = (current.items || "")
        .split(/\n|;/)
        .map((s) => s.trim())
        .filter(Boolean)
    const title = current.name && current.name.trim()
        ? current.name.trim()
        : `${dinnerLabel(curDay)} Dinner`
    const photo = current.photo && current.photo.trim() ? current.photo : ""

    function onKey(e: React.KeyboardEvent<HTMLButtonElement>, i: number) {
        let next = -1
        if (e.key === "ArrowRight" || e.key === "ArrowDown") next = (i + 1) % list.length
        else if (e.key === "ArrowLeft" || e.key === "ArrowUp") next = (i - 1 + list.length) % list.length
        else if (e.key === "Home") next = 0
        else if (e.key === "End") next = list.length - 1
        if (next < 0) return
        e.preventDefault()
        setSel(list[next].day || "")
        const el = tabRefs.current[next]
        if (el) el.focus()
    }

    let notice = ""
    if (todayKey === "Closed") {
        notice = "We're closed Mondays, so no dinner special tonight. Here's what's coming up."
    } else if (todayKey && !hasToday) {
        notice = "No dinner special today. Here's what's coming up."
    }

    return (
        <section
            id="dinner"
            className="slicers-section-pad"
            aria-labelledby="slicers-din-h"
            style={{
                background: "white",
                padding: "clamp(56px,7vw,100px) clamp(20px,5vw,80px)",
            }}
        >
            <div style={{ maxWidth: 1240, margin: "0 auto" }}>
                <V2Head
                    id="slicers-din-h"
                    num="04"
                    eyebrow="$29 dinner specials · Feeds 3–4"
                    title="Dinner is"
                    accent="handled."
                    sub="Pick a night. We do the cooking. You take the credit."
                />

                {notice ? (
                    <div
                        role="status"
                        style={{
                            marginTop: 24,
                            display: "inline-flex",
                            alignItems: "center",
                            gap: 10,
                            fontFamily: SANS,
                            fontSize: 13.5,
                            fontWeight: 600,
                            color: INK,
                            background: CREAM,
                            border: `1px solid ${TAN}`,
                            borderRadius: 999,
                            padding: "9px 16px",
                        }}
                    >
                        <span
                            aria-hidden="true"
                            style={{ width: 7, height: 7, borderRadius: "50%", background: AVAIL_CLAY, flexShrink: 0 }}
                        />
                        {notice}
                    </div>
                ) : null}

                {/* Day tabs */}
                <div className="slicers-din-tabs-wrap" style={{ marginTop: 28, maxWidth: "100%", overflowX: "auto" }}>
                    <div
                        role="tablist"
                        aria-label="Dinner special by day"
                        className="slicers-din-tabs"
                        style={{
                            display: "inline-flex",
                            gap: 8,
                            padding: 6,
                            background: CREAM,
                            border: `1px solid ${TAN}`,
                            borderRadius: 999,
                        }}
                    >
                        {list.map((d, i) => {
                            const day = d.day || ""
                            const selected = day === curDay
                            return (
                                <button
                                    key={day}
                                    ref={(el) => {
                                        tabRefs.current[i] = el
                                    }}
                                    role="tab"
                                    id={`slicers-din-tab-${i}`}
                                    aria-selected={selected}
                                    aria-controls="slicers-din-panel"
                                    tabIndex={selected ? 0 : -1}
                                    onClick={() => setSel(day)}
                                    onKeyDown={(e) => onKey(e, i)}
                                    className="slicers-din-tab"
                                    style={{
                                        position: "relative",
                                        minHeight: 46,
                                        padding: "0 18px",
                                        border: "none",
                                        borderRadius: 999,
                                        cursor: "pointer",
                                        whiteSpace: "nowrap",
                                        fontFamily: SANS,
                                        fontSize: 13,
                                        fontWeight: 700,
                                        letterSpacing: "0.06em",
                                        background: selected ? INK : "transparent",
                                        color: selected ? CREAM : INK,
                                        transition: "background-color .2s ease, color .2s ease",
                                    }}
                                >
                                    {day === "Sat / Sun" ? "Sat & Sun" : day}
                                    {day === todayKey ? (
                                        <span
                                            style={{
                                                marginLeft: 8,
                                                fontSize: 9.5,
                                                fontWeight: 800,
                                                letterSpacing: "0.14em",
                                                textTransform: "uppercase",
                                                color: selected ? "#F0907F" : RED,
                                            }}
                                        >
                                            Today
                                        </span>
                                    ) : null}
                                </button>
                            )
                        })}
                    </div>
                </div>

                {/* Meal panel */}
                <div
                    id="slicers-din-panel"
                    role="tabpanel"
                    aria-labelledby={`slicers-din-tab-${list.findIndex((d) => d.day === curDay)}`}
                    tabIndex={0}
                    className="slicers-din-panel"
                    style={{
                        marginTop: 24,
                        display: "grid",
                        gridTemplateColumns: "minmax(0,0.9fr) minmax(0,1.1fr)",
                        gap: "clamp(24px,4vw,56px)",
                        alignItems: "stretch",
                    }}
                >
                    {/* Left: photo with price stamp, or a price poster */}
                    {photo ? (
                        <div style={{ position: "relative", minHeight: 320 }}>
                            <img
                                key={photo}
                                src={photo}
                                alt={`${title} from Slicers Deli`}
                                width={900}
                                height={900}
                                loading="lazy"
                                decoding="async"
                                className="slicers-v2-fadein"
                                style={{
                                    position: "absolute",
                                    inset: 0,
                                    width: "100%",
                                    height: "100%",
                                    objectFit: "cover",
                                    borderRadius: 4,
                                    background: TAN,
                                }}
                            />
                            <div
                                aria-hidden="true"
                                style={{
                                    position: "absolute",
                                    right: 16,
                                    bottom: 16,
                                    background: RED,
                                    color: CREAM,
                                    borderRadius: 4,
                                    padding: "10px 14px",
                                    fontFamily: SLAB,
                                    fontWeight: 900,
                                    fontSize: 34,
                                    lineHeight: 1,
                                }}
                            >
                                $29
                            </div>
                        </div>
                    ) : (
                        <div
                            className="slicers-din-poster"
                            style={{
                                background: INK,
                                color: CREAM,
                                borderRadius: 4,
                                padding: "clamp(28px,4vw,44px)",
                                display: "flex",
                                flexDirection: "column",
                                justifyContent: "space-between",
                                gap: 24,
                                minHeight: 300,
                                boxSizing: "border-box",
                            }}
                        >
                            <div
                                style={{
                                    fontFamily: SANS,
                                    fontSize: 11,
                                    fontWeight: 700,
                                    letterSpacing: "0.22em",
                                    textTransform: "uppercase",
                                    color: "rgba(250,246,238,.55)",
                                }}
                            >
                                {dinnerLabel(curDay)} dinner
                            </div>
                            <div>
                                <div
                                    style={{
                                        fontFamily: SLAB,
                                        fontWeight: 900,
                                        fontSize: "clamp(88px,11vw,150px)",
                                        lineHeight: 0.85,
                                        letterSpacing: "-.04em",
                                        color: CREAM,
                                    }}
                                >
                                    $29
                                </div>
                                <div

                                fontFamily: SLAB,

// Auto-advances every 9s unless paused, hovered, focused, or the visitor
// prefers reduced motion. No star ratings (none are verified per review).
// ─────────────────────────────────────────────
function ReviewsV2({
    googleUrl,
    doordashUrl,
}: {
    googleUrl: string
    doordashUrl: string
}) {
    const list = React.useMemo(
        () => REVIEWS.filter((r) => V2_REVIEW_HOLDS.indexOf(r.name) < 0),
        []
    )
    const phone = useIsPhone()
    const reduced = usePrefersReducedMotion()
    const per = phone ? 1 : 3
    const pages = Math.max(1, Math.ceil(list.length / per))
    const [page, setPage] = React.useState(0)
    const [paused, setPaused] = React.useState(false)
    const [holding, setHolding] = React.useState(false) // hover or focus inside

    React.useEffect(() => {
        if (page >= pages) setPage(0)
    }, [pages])

    React.useEffect(() => {
        if (paused || holding || reduced || pages <= 1) return
        const id = window.setInterval(() => setPage((p) => (p + 1) % pages), 9000)
        return () => window.clearInterval(id)
    }, [paused, holding, reduced, pages])

    const shown = list.slice(page * per, page * per + per)
    const ctrl: React.CSSProperties = {
        width: 46,
        height: 46,
        borderRadius: "50%",
        border: `1.5px solid ${INK}`,
        background: "transparent",
        color: INK,
        cursor: "pointer",
        fontFamily: SANS,
        fontSize: 18,
        fontWeight: 700,
        display: "inline-flex",
        alignItems: "center",
        justifyContent: "center",
    }

    return (
        <section
            id="reviews"
            className="slicers-section-pad"
            aria-labelledby="slicers-rev-h"
            style={{
                background: CREAM,
                padding: "clamp(56px,7vw,100px) clamp(20px,5vw,80px)",

                    />
                    <div style={{ display: "flex", gap: 10, alignItems: "center" }}>
                        <button
                            type="button"
                            aria-label="Previous reviews"
                            onClick={() => setPage((p) => (p - 1 + pages) % pages)}
                            style={ctrl}
                        >
                            ‹
                        </button>
                        <button
                            type="button"
                            aria-label={paused ? "Resume auto-advance" : "Pause auto-advance"}
                            aria-pressed={paused}
                            onClick={() => setPaused(!paused)}
                            style={{ ...ctrl, fontSize: 12 }}

                            {paused ? "▶" : "❚❚"}
                        </button>
                        <button
                            type="button"
                            aria-label="Next reviews"
                            onClick={() => setPage((p) => (p + 1) % pages)}
                            style={ctrl}
                        >
                            ›
                        </button>
                    </div>
                </div>

                <div
                    aria-live={paused || reduced ? "polite" : "off"}
                    onMouseEnter={() => setHolding(true)}
                    onMouseLeave={() => setHolding(false)}
                    onFocus={() => setHolding(true)}
                    onBlur={() => setHolding(false)}
                    style={{
                        marginTop: "clamp(28px,4vw,44px)",
                        display: "grid",

                        gap: "clamp(24px,3vw,48px)",
